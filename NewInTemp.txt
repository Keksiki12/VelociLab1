name: Custom.Detection.NewInTemp
description: Обнаружение новых исполняемых файлов в TEMP-каталогах Windows и Linux
author: lab
type: CLIENT

sources:
  - name: NewInTemp
    query: |
      SELECT OSPath AS FullPath,
             Size,
             Mtime,
             hash(path=OSPath).MD5 AS MD5
      FROM glob(globs=[
        "C:/Users/user/AppData/Local/Temp/**",
        "C:/Windows/Temp/**",
        "/tmp/**",
        "/var/tmp/**"
      ])
      WHERE NOT IsDir
        AND Mtime > timestamp(epoch=now() - 86400)
        AND (
          OSPath =~ "(?i)(\\.exe|\\.dll|\\.ps1|\\.bat|\\.vbs)$"
          OR (OSPath =~ "^/" AND Size > 102400)
        )
