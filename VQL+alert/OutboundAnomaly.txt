name: Custom.Detection.OutboundAnomaly
description: Обнаружение исходящих соединений (Windows + Linux) с алертом
author: lab
type: CLIENT

required_permissions:
  - EXECVE

sources:
  # Источник для Windows (PowerShell)
  - name: Windows
    precondition: SELECT OS FROM info() WHERE OS = 'windows'
    query: |
      LET ps_script = "Get-NetTCPConnection -State Established | Where-Object { $_.RemotePort -ne 80 -and $_.RemotePort -ne 443 -and $_.RemotePort -ne 53 -and $_.RemotePort -ne 3389 } | Select-Object LocalAddress, LocalPort, RemoteAddress, RemotePort, State, OwningProcess | ConvertTo-Json -Compress"
      
      LET raw = SELECT Stdout
        FROM execve(argv=[
          "powershell.exe", "-NoProfile", "-EncodedCommand",
          base64encode(string=utf16_encode(string=ps_script))
        ], length=1000000)
        WHERE Stdout

      SELECT get(item=Parse, field="OwningProcess") AS Pid,
             get(item=Parse, field="RemoteAddress") AS RemoteIP,
             get(item=Parse, field="RemotePort") AS RemotePort,
             get(item=Parse, field="LocalAddress") AS LocalIP,
             get(item=Parse, field="LocalPort") AS LocalPort,
             get(item=Parse, field="State") AS Status,
             alert(
               name=format(
                 format="Outbound anomaly: %v:%v from PID %v",
                 args=(get(item=Parse, field="RemoteAddress"),
                       get(item=Parse, field="RemotePort"),
                       get(item=Parse, field="OwningProcess"))),
               dedup=-1,
               Level="HIGH",
               Pid=get(item=Parse, field="OwningProcess"),
               RemoteIP=get(item=Parse, field="RemoteAddress"),
               RemotePort=get(item=Parse, field="RemotePort"),
               LocalIP=get(item=Parse, field="LocalAddress"),
               LocalPort=get(item=Parse, field="LocalPort"),
               Status=get(item=Parse, field="State")
             ) AS AlertSent
      FROM foreach(row=raw, query={ SELECT parse_json_array(data=Stdout) AS Parse FROM scope() })
      WHERE RemoteIP AND RemotePort

  # Источник для Linux (netstat)
  - name: Linux
    precondition: SELECT OS FROM info() WHERE OS = 'linux'
    query: |
      SELECT Pid,
             process_tracker_get(id=Pid).Name AS ProcessName,
             process_tracker_get(id=Pid).Exe AS Exe,
             Laddr.IP AS LocalIP, Laddr.Port AS LocalPort,
             Raddr.IP AS RemoteIP, Raddr.Port AS RemotePort,
             Status,
             alert(
               name=format(
                 format="Outbound anomaly: %v:%v from %v",
                 args=(Raddr.IP, Raddr.Port,
                       process_tracker_get(id=Pid).Name)),
               dedup=-1,
               Level="HIGH",
               Pid=Pid,
               ProcessName=process_tracker_get(id=Pid).Name,
               Exe=process_tracker_get(id=Pid).Exe,
               RemoteIP=Raddr.IP,
               RemotePort=Raddr.Port,
               LocalIP=Laddr.IP,
               LocalPort=Laddr.Port,
               Status=Status
             ) AS AlertSent
      FROM netstat()
      WHERE Status = "ESTABLISHED"
        AND NOT (Raddr.Port IN (80, 443, 53, 22, 3389))
        AND Raddr.IP != "127.0.0.1"
