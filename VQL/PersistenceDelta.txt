name: Custom.Detection.PersistenceDelta
description: Обнаружение записей автозапуска (Windows + Linux)
author: lab
type: CLIENT

required_permissions:
  - EXECVE

sources:
  # Источник для Windows (Реестр через PowerShell)
  - name: Windows
    precondition: SELECT OS FROM info() WHERE OS = 'windows'
    query: |
      LET ps_script = "$path = 'HKLM:\\SOFTWARE\\Microsoft\\Windows\\CurrentVersion\\Run'; $items = Get-ItemProperty -Path $path; $items.PSObject.Properties | Where-Object { $_.Name -notlike 'PS*' } | ForEach-Object { [PSCustomObject]@{ Name = $_.Name; Value = $_.Value } } | ConvertTo-Json -Compress"
      
      LET raw = SELECT Stdout
        FROM execve(argv=[
          "powershell.exe", "-NoProfile", "-EncodedCommand",
          base64encode(string=utf16_encode(string=ps_script))
        ], length=1000000)
        WHERE Stdout

      SELECT get(item=Parse, field="Name") AS ValueName,
             get(item=Parse, field="Value") AS Value,
             "HKEY_LOCAL_MACHINE\\SOFTWARE\\Microsoft\\Windows\\CurrentVersion\\Run" AS RegistryKey
      FROM foreach(row=raw, query={ SELECT parse_json_array(data=Stdout) AS Parse FROM scope() })
      WHERE ValueName =~ "(?i)(BenignPersistence|benign_attack)"

  # Источник для Linux (cron/systemd)
  - name: Linux
    precondition: SELECT OS FROM info() WHERE OS = 'linux'
    query: |
      SELECT OSPath AS FilePath, Mtime, read_file(filename=OSPath) AS Content
      FROM glob(globs=[
        "/etc/crontab", "/etc/cron.d/**", "/var/spool/cron/crontabs/**",
        "/etc/systemd/system/**", "/etc/rc.local", "/etc/init.d/**",
        "/home/*/.config/autostart/**"
      ])
      WHERE NOT IsDir
        AND Mtime > timestamp(epoch=now() - 604800)
        AND Content =~ "(?i)(benign|attack|reverse|4444)"
